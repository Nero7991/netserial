cmake_minimum_required(VERSION 3.16)
project(ns_tests C)
set(CMAKE_C_STANDARD 99)
file(GLOB CORE_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../core/*.c)
file(GLOB TEST_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test_*.c)
add_executable(ns_tests ${TEST_SRCS} ${CORE_SRCS})
target_include_directories(ns_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../core ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(ns_tests PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(ns_tests PRIVATE -fsanitize=address,undefined)
